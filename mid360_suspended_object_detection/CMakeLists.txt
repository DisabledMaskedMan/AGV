cmake_minimum_required(VERSION 3.8)
project(mid360_suspended_object_detection)

option(BUILD_ROS2 "Build the existing ROS 2 node" ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Boost 1.75 REQUIRED COMPONENTS system json)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(scan_core STATIC src/scan_core.cpp src/scan_config_store.cpp src/publication_telemetry.cpp)
target_compile_features(scan_core PUBLIC cxx_std_17)
target_include_directories(scan_core PUBLIC src)
target_link_libraries(scan_core PUBLIC Boost::json yaml-cpp Threads::Threads)
target_compile_options(scan_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(scan_web_server src/scan_web_server.cpp)
target_link_libraries(scan_web_server scan_core Boost::system)
target_compile_options(scan_web_server PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(scan_web_server PRIVATE SCAN_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
install(TARGETS scan_web_server DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scan_web/static DESTINATION share/${PROJECT_NAME}/scan_web)
include(CTest)
if(BUILD_TESTING)
  add_executable(scan_core_test tests/scan_core_test.cpp)
  target_link_libraries(scan_core_test PRIVATE scan_core)
  add_test(NAME scan_core_test COMMAND scan_core_test)
  add_executable(config_store_test tests/config_store_test.cpp)
  target_link_libraries(config_store_test PRIVATE scan_core)
  add_test(NAME config_store_test COMMAND config_store_test)
  add_executable(publication_test tests/publication_test.cpp)
  target_link_libraries(publication_test PRIVATE scan_core)
  add_test(NAME publication_test COMMAND publication_test)
  add_executable(scan_fixture tests/scan_fixture.cpp)
  target_link_libraries(scan_fixture PRIVATE scan_core)
endif()

if(BUILD_ROS2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(rosidl_default_generators REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/LargestObstacle.msg"
  DEPENDENCIES std_msgs geometry_msgs
)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
ament_export_dependencies(rosidl_default_runtime)

target_compile_definitions(scan_web_server PRIVATE SCAN_WITH_ROS2=1)
ament_target_dependencies(scan_web_server rclcpp std_msgs geometry_msgs)
target_link_libraries(scan_web_server "${cpp_typesupport_target}")

add_executable(suspended_object_node src/suspended_object_node.cpp)
target_compile_features(suspended_object_node PRIVATE cxx_std_17)
target_compile_options(suspended_object_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(suspended_object_node rclcpp std_msgs visualization_msgs livox_ros_driver2)
install(TARGETS suspended_object_node DESTINATION lib/${PROJECT_NAME})
endif()
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_ROS2)
  install(DIRECTORY msg DESTINATION share/${PROJECT_NAME})
endif()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_ROS2)
  ament_package()
endif()
